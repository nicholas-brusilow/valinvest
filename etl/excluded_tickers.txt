# Rule: a ticker is excluded when its split-adjusted quarterly close
# (price_quarterly.close_raw) moves more than 100x up, or less than 1/100
# down, between consecutive quarters.  Such jumps are treated as corrupt
# vendor data and the ticker is removed from the dataset entirely.
#
# Generated: 2026-09-17
#
# Authoritative audit query:
#   WITH x AS (
#     SELECT ticker_yahoo, quarter, close_raw,
#            LAG(close_raw) OVER (PARTITION BY ticker_yahoo ORDER BY quarter) AS prev
#     FROM price_quarterly WHERE close_raw IS NOT NULL AND close_raw > 0
#   )
#   SELECT DISTINCT ticker_yahoo FROM x
#   WHERE prev > 0 AND (close_raw / prev > 100 OR close_raw / prev < 0.01)
#   ORDER BY ticker_yahoo;
#
# Applied by both ETLs (etl/common.py:excluded_tickers) and
# sql/postprocess.sql (D3), so excluded names are never re-ingested.
